4. Компоненти

Кнопка (.btn, .btn--primary)

<!-- Стандартна кнопка -->
<button class="btn">Кнопка</button>
<a href="#" class="btn">Посилання-кнопка</a>

<!-- Акцентна кнопка (редизайн) -->
<button class="btn btn--primary">Почати</button>

<!-- Вимкнений стан -->
<button class="btn is-disabled" disabled>Вимкнена</button>
